cmake_minimum_required(VERSION 3.27)
project(snow_media_native LANGUAGES C CXX)
if(APPLE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum macOS version" FORCE)
endif()
option(SNOW_MEDIA_BUILD_RECORDING "Build FFmpeg recording/export C library" OFF)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/RustStaticLibrary.cmake")
snow_add_rust_static_library(snow_capture_c PACKAGE snow-capture-c
    MANIFEST_DIR "${CMAKE_CURRENT_LIST_DIR}/..")
target_include_directories(snow_capture_c INTERFACE "${CMAKE_CURRENT_LIST_DIR}/../crates/snow-capture-c/include")
if(APPLE)
    foreach(_framework AppKit CoreText Carbon CloudKit CoreData Foundation CoreFoundation CoreGraphics CoreMedia CoreVideo
        ScreenCaptureKit AVFoundation AVFAudio AudioToolbox CoreAudio Metal CoreImage
        QuartzCore IOSurface ImageIO UniformTypeIdentifiers MediaToolbox)
        target_link_libraries(snow_capture_c INTERFACE "-framework ${_framework}")
    endforeach()
    target_link_libraries(snow_capture_c INTERFACE objc iconv c++)
    add_executable(snow_capture_c_smoke c_capture_smoke.c)
    target_link_libraries(snow_capture_c_smoke PRIVATE snow_capture_c)
    enable_testing()
    add_test(NAME snow-capture-c-macos-abi COMMAND snow_capture_c_smoke)
endif()
if(SNOW_MEDIA_BUILD_RECORDING)
    snow_add_rust_static_library(snow_recording_c PACKAGE snow-recording-c
        MANIFEST_DIR "${CMAKE_CURRENT_LIST_DIR}/..")
    target_include_directories(snow_recording_c INTERFACE "${CMAKE_CURRENT_LIST_DIR}/../crates/snow-recording-c/include")
    find_package(PkgConfig REQUIRED)
    if(DEFINED ENV{FFMPEG_DIR})
        set(ENV{PKG_CONFIG_PATH} "$ENV{FFMPEG_DIR}/lib/pkgconfig")
    endif()
    pkg_check_modules(SNOW_MEDIA_FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavformat libavutil libswscale libswresample)
    target_link_libraries(snow_recording_c INTERFACE snow_capture_c PkgConfig::SNOW_MEDIA_FFMPEG)
    if(APPLE)
        add_executable(snow_recording_c_smoke c_recording_smoke.c)
        target_link_libraries(snow_recording_c_smoke PRIVATE snow_recording_c)
        add_test(NAME snow-recording-c-macos-abi COMMAND snow_recording_c_smoke)
    endif()
endif()
