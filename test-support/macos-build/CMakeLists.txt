cmake_minimum_required(VERSION 4.2)
project(snow_macos_fixture VERSION 1.0.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OSX_DEPLOYMENT_TARGET 15.0)
set(SNOW_SHOT_VERSION 1.0.5-beta)
set(SNOW_SHOT_VERSION_NUMERIC 1.0.5)
set(SNOW_SHOT_VENDOR "Snow Apps")
set(SNOW_SHOT_COPYRIGHT "Copyright (C) 2025-2026 mg-chao")
# This fixture provisions the shared Qt/ONNX runtime deployment path.
set(SNOW_SHOT_RELEASE_STATIC OFF)
set(SNOW_SHOT_QT_STATIC OFF)
set(SNOW_SHOT_OCR_STATIC_ONNXRUNTIME OFF)
if(NOT SNOW_FFMPEG_ROOT)
    message(FATAL_ERROR "Pass SNOW_FFMPEG_ROOT to the provisioned native FFmpeg prefix")
endif()
list(APPEND CMAKE_MODULE_PATH "${SNOW_ROOT}/cmake")
find_package(Qt6 6.11.1 EXACT REQUIRED COMPONENTS Widgets)
add_library(snow_capture_c INTERFACE)
add_library(snow_recording_c INTERFACE)
add_library(onnxruntime SHARED dummy.cpp)
add_library(onnxruntime::onnxruntime ALIAS onnxruntime)
set_target_properties(onnxruntime PROPERTIES VERSION 1.28.0 SOVERSION 1)
add_library(snow_shot_image_codec_backend SHARED dummy.cpp)
add_executable(snow_shot MACOSX_BUNDLE main.cpp)
target_link_libraries(snow_shot PRIVATE Qt6::Widgets snow_recording_c snow_shot_image_codec_backend)
add_dependencies(snow_shot onnxruntime)
add_executable(snow-ocr-process helper.cpp)
set(CMAKE_INSTALL_DEFAULT_COMPONENT_NAME SnowShot)
install(TARGETS snow_shot BUNDLE DESTINATION .)
install(TARGETS snow_shot_image_codec_backend
    LIBRARY DESTINATION snow_shot.app/Contents/Frameworks)
install(TARGETS snow-ocr-process RUNTIME DESTINATION snow_shot.app/Contents/MacOS)
find_program(SNOW_CRASHPAD_HANDLER NAMES crashpad_handler
    PATHS "${SNOW_FFMPEG_ROOT}/tools/crashpad" "${SNOW_FFMPEG_ROOT}/tools"
    NO_DEFAULT_PATH REQUIRED)
find_library(SNOW_FIXTURE_ZLIB NAMES z PATHS "${SNOW_FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
if(NOT TARGET ZLIB::ZLIB)
    add_library(ZLIB::ZLIB UNKNOWN IMPORTED)
    set_target_properties(ZLIB::ZLIB PROPERTIES IMPORTED_LOCATION "${SNOW_FIXTURE_ZLIB}")
endif()
include(SnowShotMacOS)
target_link_libraries(onnxruntime PRIVATE PkgConfig::SNOW_SHOT_FFMPEG)
target_link_libraries(snow_shot_image_codec_backend PRIVATE PkgConfig::SNOW_SHOT_FFMPEG)
include(SnowShotMacOSPackage)
